function parseTextToQuestions(rawHtml, grade, topic, defaultLevel, selectedType) { let parsedArr = []; if (rawHtml.normalize) { rawHtml = rawHtml.normalize('NFC'); } let tempDiv = document.createElement('div'); tempDiv.innerHTML = rawHtml; // Chuyển đổi các thẻ HTML thành ký tự xuống dòng thuần túy để dễ quét dữ liệu let textContent = tempDiv.innerHTML.replace(//gi, '\n').replace(/<\/?(div|p|h[1-6]|ul|ol|li)[^>]*>/gi, '\n'); // Tách các câu hỏi dựa trên từ khóa "Câu" hoặc "Bài" kèm theo số thứ tự let blocks = textContent.split(/(?=(?:C[ââ]u|B[àà]i)\s*\d+[\s:.\-_])/i); blocks.forEach((blockStr) => { let block = blockStr.trim(); if (!block) return; const qId = `CH${grade}_${topic}_${Date.now().toString().slice(-4)}_${parsedArr.length + 1}`; let options = ['', '', '', '']; let ans = 'A'; let ansOptions = []; let currentType = selectedType; if (/a[\)\.]\s/i.test(block) || /b[\)\.]\s/i.test(block) || /c[\)\.]\s/i.test(block)) { currentType = 'TF'; } else if (/Đáp án\s*[:\-]?\s*[D|S]/i.test(block)) { currentType = 'TF'; } if (currentType === 'TF') { // Lấy đáp án kiểu D-S-D-S hoặc DSDS let ansMatch = block.match(/(?:Đáp án|Chọn)\s*[:\-]?\s*([D|S][-\s]*[D|S][-\s]*[D|S][-\s]*[D|S]|[D|S][-\s]*[D|S][-\s]*[D|S]|[D|S][-\s]*[D|S])/i); if (ansMatch) { ans = ansMatch[1].toUpperCase().replace(/[^DS]/g, ''); ansOptions = ans.split(''); block = block.substring(0, ansMatch.index).trim(); } else { ans = "D-S-D-S"; ansOptions = ['D', 'S', 'D', 'S']; } // Quét từng ý a, b, c, d dựa trên ký tự xuống dòng let lines = block.split('\n'); let qLines = []; lines.forEach(line => { let lTrim = line.trim(); let matchA = lTrim.match(/^a[\)\.]\s*(.*)$/i); let matchB = lTrim.match(/^b[\)\.]\s*(.*)$/i); let matchC = lTrim.match(/^c[\)\.]\s*(.*)$/i); let matchD = lTrim.match(/^d[\)\.]\s*(.*)$/i); if (matchA) options[0] = matchA[1].trim(); else if (matchB) options[1] = matchB[1].trim(); else if (matchC) options[2] = matchC[1].trim(); else if (matchD) options[3] = matchD[1].trim(); else { qLines.push(lTrim); } }); let qContent = qLines.join('
').replace(/^(?:C[ââ]u|B[àà]i)\s*\d+[\s:.\-_]+/i, '').trim(); parsedArr.push({ id: qId, grade: grade, topic: topic, level: defaultLevel, type: 'TF', q: qContent, options: options, ans: ans, ansOptions: ansOptions }); } else if (currentType === 'SA') { let ansMatch = block.match(/(?:Đáp án|Chọn)\s*[:\-]?\s*(.*?)$/i); if (ansMatch) { ans = ansMatch[1].trim(); block = block.substring(0, ansMatch.index).trim(); } let qContent = block.replace(/^(?:C[ââ]u|B[àà]i)\s*\d+[\s:.\-_]+/i, '').trim().replace(/\n/g, '
'); parsedArr.push({ id: qId, grade: grade, topic: topic, level: defaultLevel, type: 'SA', q: qContent, options: [], ans: ans }); } else { let ansMatch = block.match(/(?:Đáp án|Chọn)\s*[:\-]?\s*([A-D])/i); if (ansMatch) { ans = ansMatch[1].toUpperCase(); block = block.substring(0, ansMatch.index).trim(); } let lines = block.split('\n'); let qLines = []; lines.forEach(line => { let lTrim = line.trim(); let mA = lTrim.match(/^A[\)\.]\s*(.*)$/i); let mB = lTrim.match(/^B[\)\.]\s*(.*)$/i); let mC = lTrim.match(/^C[\)\.]\s*(.*)$/i); let mD = lTrim.match(/^D[\)\.]\s*(.*)$/i); if (mA) options[0] = mA[1].trim(); else if (mB) options[1] = mB[1].trim(); else if (mC) options[2] = mC[1].trim(); else if (mD) options[3] = mD[1].trim(); else qLines.push(lTrim); }); let qContent = qLines.join('
').replace(/^(?:C[ââ]u|B[àà]i)\s*\d+[\s:.\-_]+/i, '').trim(); parsedArr.push({ id: qId, grade: grade, topic: topic, level: defaultLevel, type: 'MCQ', q: qContent, options: options, ans: ans }); } }); return parsedArr; }
