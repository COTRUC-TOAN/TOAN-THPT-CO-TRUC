const TEACHER_PASSWORD = "CoTruc@2026"; function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const action = e && e.parameter ? e.parameter.action : ""; // 1. LẤY DỮ LIỆU SỔ ĐIỂM if (action === "get_results") { if (e.parameter.password !== TEACHER_PASSWORD) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sai mật khẩu!" })) .setMimeType(ContentService.MimeType.JSON); } const sheetKQ = ss.getSheetByName("KetQua"); const dataKQ = sheetKQ ? sheetKQ.getDataRange().getValues() : []; const results = []; for (let i = 1; i < dataKQ.length; i++) { const [time, examId, examTitle, name, sClass, mode, p1, p2, p3, total] = dataKQ[i]; if (!name) continue; results.push({ time: time ? time.toString() : "", examId: (examId || "").toString(), examTitle: (examTitle || "").toString(), name: (name || "").toString(), class: (sClass || "").toString().trim().toUpperCase(), mode: (mode || "").toString(), p1Score: p1, p2Score: p2, p3Score: p3, totalScore: total }); } return ContentService.createTextOutput(JSON.stringify({ status: "success", results: results })) .setMimeType(ContentService.MimeType.JSON); } // 2. LẤY NGÂN HÀNG CÂU HỎI if (action === "get_question_bank") { if (e.parameter.password !== TEACHER_PASSWORD) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sai mật khẩu!" })) .setMimeType(ContentService.MimeType.JSON); } const sheetCH = ss.getSheetByName("NganHangCauHoi"); const dataCH = sheetCH ? sheetCH.getDataRange().getValues() : []; const qList = []; for (let i = 1; i < dataCH.length; i++) { const [id, grade, topic, level, type, q, a, b, c, d, ans, exp] = dataCH[i]; if (!id || !q) continue; qList.push({ id: (id || "").toString(), grade: (grade || "").toString().trim(), topic: (topic || "").toString().trim(), level: (level || "NB").toString().trim().toUpperCase(), type: (type || "MCQ").toString().trim().toUpperCase(), q: (q || "").toString().trim(), options: [ (a || "").toString().trim(), (b || "").toString().trim(), (c || "").toString().trim(), (d || "").toString().trim() ], ans: (ans || "").toString().trim(), explanation: (exp || "").toString().trim() }); } return ContentService.createTextOutput(JSON.stringify({ status: "success", questions: qList })) .setMimeType(ContentService.MimeType.JSON); } // 3. TRẢ VỀ ĐỀ CHO HỌC SINH const sheetDe = ss.getSheetByName("NganHangDe"); const dataDe = sheetDe ? sheetDe.getDataRange().getValues() : []; const examsMap = {}; for (let i = 1; i < dataDe.length; i++) { const [maDe, tenDe, khoi, chuyenDe, cheDo, thoiGian, loaiCau, cauHoi, a, b, c, d, dAn, lGiai] = dataDe[i]; if (!maDe || !cauHoi) continue; const key = maDe.toString().trim(); const mode = (cheDo || "LUYENTAP").toString().trim().toUpperCase(); if (!examsMap[key]) { examsMap[key] = { id: key, title: (tenDe || "").toString().trim(), grade: (khoi || "").toString().trim().toLowerCase(), topic: (chuyenDe || "").toString().trim(), mode: mode, timeLimit: parseInt(thoiGian) || 15, questions: [] }; } const qObj = { type: (loaiCau || "MCQ").toString().trim().toUpperCase(), q: (cauHoi || "").toString().trim(), options: [ (a || "").toString().trim(), (b || "").toString().trim(), (c || "").toString().trim(), (d || "").toString().trim() ] }; if (mode === "LUYENTAP") { qObj.ans = (dAn || "").toString().trim(); qObj.explanation = (lGiai || "").toString().trim(); } examsMap[key].questions.push(qObj); } // 4. LẤY LÝ THUYẾT const sheetLT = ss.getSheetByName("LyThuyet"); const dataLT = sheetLT ? sheetLT.getDataRange().getValues() : []; const theoriesMap = {}; for (let i = 1; i < dataLT.length; i++) { const [chuyenDe, tieuDe, linkAnh, noiDung] = dataLT[i]; if (!chuyenDe) continue; theoriesMap[chuyenDe.toString().trim()] = { topic: chuyenDe.toString().trim(), title: (tieuDe || "").toString().trim(), image: (linkAnh || "").toString().trim(), content: (noiDung || "").toString().trim() }; } return ContentService.createTextOutput(JSON.stringify({ status: "success", exams: Object.values(examsMap), theories: theoriesMap })).setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const postData = JSON.parse(e.postData.contents); const action = postData.action || "submit_quiz"; // A. LƯU ĐỀ MỚI SOẠN TỪ WEB VÀO TAB NGANHANGDE if (action === "save_exam") { if (postData.password !== TEACHER_PASSWORD) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sai mật khẩu quản lý!" })) .setMimeType(ContentService.MimeType.JSON); } const sheetDe = ss.getSheetByName("NganHangDe"); const questions = postData.questions || []; questions.forEach(q => { sheetDe.appendRow([ postData.examId, postData.examTitle, postData.grade, postData.topic, postData.mode, postData.timeLimit, q.type, q.q, q.options[0] || "", q.options[1] || "", q.options[2] || "", q.options[3] || "", q.ans || "", q.explanation || "" ]); }); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Đã lưu đề mới thành công!" })) .setMimeType(ContentService.MimeType.JSON); } // B. THÊM CÂU HỎI MỚI VÀO TAB NGANHANGCAUHOI if (action === "add_question") { if (postData.password !== TEACHER_PASSWORD) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Sai mật khẩu!" })) .setMimeType(ContentService.MimeType.JSON); } const sheetCH = ss.getSheetByName("NganHangCauHoi"); const q = postData.question; sheetCH.appendRow([ q.id, q.grade, q.topic, q.level, q.type, q.q, q.options[0] || "", q.options[1] || "", q.options[2] || "", q.options[3] || "", q.ans || "", q.explanation || "" ]); return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Đã thêm câu hỏi vào Ngân hàng!" })) .setMimeType(ContentService.MimeType.JSON); } // C. CHẤM ĐIỂM BÀI THI if (action === "submit_quiz") { const sheetDe = ss.getSheetByName("NganHangDe"); const dataDe = sheetDe ? sheetDe.getDataRange().getValues() : []; const examId = postData.examId; const correctAnswers = []; for (let i = 1; i < dataDe.length; i++) { if (dataDe[i][0].toString().trim() === examId) { correctAnswers.push({ type: (dataDe[i][6] || "MCQ").toString().trim().toUpperCase(), ans: (dataDe[i][12] || "").toString().trim(), explanation: (dataDe[i][13] || "").toString().trim() }); } } let p1Score = 0.0, p2Score = 0.0, p3Score = 0.0; const userAnswers = postData.answers || []; for (let i = 0; i < correctAnswers.length; i++) { const qType = correctAnswers[i].type; const rightAns = correctAnswers[i].ans; const uAns = userAnswers[i]; if (!uAns) continue; if (qType === "MCQ") { if (uAns.toString().trim().toUpperCase() === rightAns.toUpperCase()) p1Score += 0.25; } else if (qType === "TF") { const cleanRight = rightAns.replace(/[^DSds]/g, "").toUpperCase(); let cCount = 0; ['a', 'b', 'c', 'd'].forEach((ch, idx) => { if (uAns[ch] && uAns[ch].toString().toUpperCase() === cleanRight[idx]) cCount++; }); if (cCount === 1) p2Score += 0.1; else if (cCount === 2) p2Score += 0.25; else if (cCount === 3) p2Score += 0.5; else if (cCount === 4) p2Score += 1.0; } else if (qType === "SA") { const valStr = uAns.toString().replace(/\s+/g, "").replace(",", "."); const validList = rightAns.split(/[|;]/).map(a => a.trim().replace(/\s+/g, "").replace(",", ".")); if (validList.includes(valStr) && valStr !== "") p3Score += 0.5; } } const totalScore = (p1Score + p2Score + p3Score).toFixed(2); let sheetKQ = ss.getSheetByName("KetQua"); if (!sheetKQ) { sheetKQ = ss.insertSheet("KetQua"); sheetKQ.appendRow(["ThoiGian", "MaDe", "TenDe", "HoTen", "Lop", "CheDo", "Diem_P1", "Diem_P2", "Diem_P3", "TongDiem"]); } sheetKQ.appendRow([ postData.time, postData.examId, postData.examTitle, postData.name, postData.class, postData.mode, p1Score.toFixed(2), p2Score.toFixed(2), p3Score.toFixed(2), totalScore ]); return ContentService.createTextOutput(JSON.stringify({ status: "success", p1Score: p1Score.toFixed(2), p2Score: p2Score.toFixed(2), p3Score: p3Score.toFixed(2), totalScore: totalScore, details: postData.mode === "LUYENTAP" ? correctAnswers : [] })).setMimeType(ContentService.MimeType.JSON); } } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
